[TestClass]
[DoNotParallelize]
public sealed class QueueTests : IRegressionTestMetadata
{
    public IReadOnlyDictionary<string, RegressionFeature> RegressionFeatures { get; } =
        new Dictionary<string, RegressionFeature>
        {
            ["probe-availability"] = new("Queueing & Scheduling", "Probe availability", "Keeps health probes responsive without allowing ordinary traffic to consume reserved probe capacity."),
            ["retry-admission"] = new("Queueing & Scheduling", "Retry admission", "Ensures retryable work is not dropped when the admission limit for new work has been reached."),
            ["priority-ordering"] = new("Queueing & Scheduling", "Priority ordering", "Ensures urgent and preferred-priority requests reach the intended workers before lower-priority work."),
            ["concurrent-delivery"] = new("Queueing & Scheduling", "Concurrent delivery", "Guards against lost or duplicate requests when many priority workers process the queue concurrently.")
        };

    [TestMethod]
    [RegressionTestCase("probe-availability", "Queued probes resume when the probe worker is available", "A probe waiting in the queue must be delivered as soon as reserved probe capacity becomes available.")]
    public async Task QueuedProbe_IsDeliveredWhenProbeWorkerBecomesAvailable()
    {
        var queue = CreateQueue();
        using var cancellationSource = new CancellationTokenSource();
        var signalerTask = queue.SignalWorker(cancellationSource.Token);

        try
        {
            var firstProbe = Guid.NewGuid();
            var firstProbeWait = queue.DequeueAsync(preferredPriority: 0);
            Assert.IsTrue(queue.Enqueue(firstProbe, priority: 0, priority2: 0, DateTime.UtcNow));
            Assert.AreEqual(firstProbe, await firstProbeWait.WaitAsync(TimeSpan.FromSeconds(1)));

            var queuedProbe = Guid.NewGuid();
            Assert.IsTrue(queue.Enqueue(queuedProbe, priority: 0, priority2: 0, DateTime.UtcNow));
            Assert.AreEqual(1, queue.thrdSafeCount);

            var queuedProbeWait = queue.DequeueAsync(preferredPriority: 0);
            Assert.AreEqual(queuedProbe, await queuedProbeWait.WaitAsync(TimeSpan.FromSeconds(1)));
            Assert.AreEqual(0, queue.thrdSafeCount);
        }
        finally
        {
            await DrainAndStopAsync(queue, cancellationSource, signalerTask);
        }
    }

    [TestMethod]
    [RegressionTestCase("probe-availability", "Queued probes take precedence over ordinary work", "When a probe and ordinary request are queued together, the next available regular worker must service the probe first.")]
    public async Task QueuedProbe_IsDeliveredBeforeOrdinaryWorkToNextRegularWorker()
    {
        var queue = CreateQueue();
        using var cancellationSource = new CancellationTokenSource();
        var signalerTask = queue.SignalWorker(cancellationSource.Token);

        try
        {
            var firstProbe = Guid.NewGuid();
            var probeWorkerWait = queue.DequeueAsync(preferredPriority: 0);
            Assert.IsTrue(queue.Enqueue(firstProbe, priority: 0, priority2: 0, DateTime.UtcNow));
            Assert.AreEqual(firstProbe, await probeWorkerWait.WaitAsync(TimeSpan.FromSeconds(1)));

            var ordinaryRequest = Guid.NewGuid();
            var secondProbe = Guid.NewGuid();
            Assert.IsTrue(queue.Enqueue(ordinaryRequest, priority: 1, priority2: 0, DateTime.UtcNow));
            Assert.IsTrue(queue.Enqueue(secondProbe, priority: 0, priority2: 0, DateTime.UtcNow));

            var regularWorkerWait = queue.DequeueAsync(preferredPriority: 1);
            Assert.AreEqual(secondProbe, await regularWorkerWait.WaitAsync(TimeSpan.FromSeconds(1)));
        }
        finally
        {
            await DrainAndStopAsync(queue, cancellationSource, signalerTask);
        }
    }

    [TestMethod]
    [RegressionTestCase("probe-availability", "Waiting regular workers can service concurrent probes", "A probe arriving while a regular worker waits must be delivered immediately rather than remaining queued.")]
    public async Task ConcurrentProbe_IsDeliveredDirectlyToWaitingRegularWorker()
    {
        var queue = CreateQueue();
        using var cancellationSource = new CancellationTokenSource();
        var signalerTask = queue.SignalWorker(cancellationSource.Token);

        try
        {
            var firstProbe = Guid.NewGuid();
            var probeWorkerWait = queue.DequeueAsync(preferredPriority: 0);

            Assert.AreEqual(firstProbe, await probeWorkerWait.WaitAsync(TimeSpan.FromSeconds(1)));

            var secondProbe = Guid.NewGuid();
            var regularWorkerWait = queue.DequeueAsync(preferredPriority: 1);
            Assert.IsTrue(queue.Enqueue(secondProbe, priority: 0, priority2: 0, DateTime.UtcNow));

            Assert.AreEqual(secondProbe, await regularWorkerWait.WaitAsync(TimeSpan.FromSeconds(1)));

    [TestMethod]
    [RegressionTestCase("probe-availability", "Ordinary requests cannot consume the reserved probe worker", "Normal traffic must remain queued until a regular worker is available, preserving dedicated probe capacity.")]
    public async Task OrdinaryWork_DoesNotUseReservedProbeWorker()
    {
        var queue = CreateQueue();
        using var cancellationSource = new CancellationTokenSource();
        var signalerTask = queue.SignalWorker(cancellationSource.Token);

        try
        {
            var ordinaryRequest = Guid.NewGuid();
            Assert.IsTrue(queue.Enqueue(ordinaryRequest, priority: 1, priority2: 0, DateTime.UtcNow));
            var probeWorkerWait = queue.DequeueAsync(preferredPriority: 0);

            await Task.Delay(100);
            Assert.IsFalse(probeWorkerWait.IsCompleted);
            Assert.AreEqual(1, queue.thrdSafeCount);

            var regularWorkerWait = queue.DequeueAsync(Constants.AnyPriority);
            Assert.AreEqual(ordinaryRequest, await regularWorkerWait.WaitAsync(TimeSpan.FromSeconds(1)));
        }
        finally
        {
            await DrainAndStopAsync(queue, cancellationSource, signalerTask);
        }
    }

    [TestMethod]
    [RegressionTestCase("retry-admission", "Retries are accepted when new-work capacity is full", "A request already admitted for processing must be requeued even when the queue rejects additional new work.")]
    public async Task Requeue_IsAcceptedAboveNewWorkAdmissionLimit()
    {
        var queue = CreateQueue(maxQueueLength: 1);
        using var cancellationSource = new CancellationTokenSource();
        var signalerTask = queue.SignalWorker(cancellationSource.Token);
        try
        {
            var firstRequest = Guid.NewGuid();
            var rejectedRequest = Guid.NewGuid();
            var requeuedRequest = Guid.NewGuid();

            Assert.IsTrue(queue.Enqueue(firstRequest, priority: 1, priority2: 0, DateTime.UtcNow));
            Assert.IsFalse(queue.Enqueue(rejectedRequest, priority: 1, priority2: 0, DateTime.UtcNow));
            Assert.IsTrue(queue.Requeue(requeuedRequest, priority: 1, priority2: 0, DateTime.UtcNow));
            Assert.AreEqual(2, queue.thrdSafeCount);

            var firstWorkerWait = queue.DequeueAsync(Constants.AnyPriority);
            var secondWorkerWait = queue.DequeueAsync(Constants.AnyPriority);
            var delivered = await Task.WhenAll(firstWorkerWait, secondWorkerWait).WaitAsync(TimeSpan.FromSeconds(1));

            CollectionAssert.AreEquivalent(new[] { firstRequest, requeuedRequest }, delivered);
            Assert.AreEqual(0, queue.thrdSafeCount);
        }
        finally
        {
            await DrainAndStopAsync(queue, cancellationSource, signalerTask);
        }
    }

    [TestMethod]
    [RegressionTestCase("priority-ordering", "Priority, boost, and age determine delivery order", "Requests must be delivered by primary priority, then user boost, then oldest timestamp.")]
    public void PriorityQueue_OrdersByPrimaryPriorityBoostAndTimestamp()
    {
        var queue = new SimpleL7Proxy.Queue.PriorityQueue<string>();
        var timestamp = DateTime.UtcNow;

        queue.Enqueue(new PriorityQueueItem<string>("priority-2", priority: 2, priority2: 1, timestamp));
        queue.Enqueue(new PriorityQueueItem<string>("normal", priority: 1, priority2: 0, timestamp));
        queue.Enqueue(new PriorityQueueItem<string>("boosted-new", priority: 1, priority2: 1, timestamp.AddTicks(2)));
        queue.Enqueue(new PriorityQueueItem<string>("boosted-old", priority: 1, priority2: 1, timestamp.AddTicks(1)));

        var delivered = new[]
        {
            queue.Dequeue(Constants.AnyPriority),
            queue.Dequeue(Constants.AnyPriority),
            queue.Dequeue(Constants.AnyPriority),
            queue.Dequeue(Constants.AnyPriority)
        };

        CollectionAssert.AreEqual(
            new[] { "boosted-old", "boosted-new", "normal", "priority-2" },
            delivered);
        Assert.AreEqual(0, queue.Count);
    }

    [TestMethod]
    [RegressionTestCase("priority-ordering", "Preferred workers receive matching priority work", "A worker assigned to a priority must receive matching queued work without consuming requests from other priorities first.")]
    public async Task PreferredPriorityWorker_ReceivesMatchingQueuedRequest()
    {
        var queue = CreateQueue();
        using var cancellationSource = new CancellationTokenSource();
        var signalerTask = queue.SignalWorker(cancellationSource.Token);

        try
        {
            var priority1 = Guid.NewGuid();
            var priority2 = Guid.NewGuid();
            var priority3 = Guid.NewGuid();
            Assert.IsTrue(queue.Enqueue(priority1, priority: 1, priority2: 0, DateTime.UtcNow));
            Assert.IsTrue(queue.Enqueue(priority2, priority: 2, priority2: 0, DateTime.UtcNow));
            Assert.IsTrue(queue.Enqueue(priority3, priority: 3, priority2: 0, DateTime.UtcNow));

            var preferredWorkerWait = queue.DequeueAsync(preferredPriority: 2);

            Assert.AreEqual(priority2, await preferredWorkerWait.WaitAsync(TimeSpan.FromSeconds(1)));
            Assert.AreEqual(2, queue.thrdSafeCount);
        }
        finally
        {
            await DrainAndStopAsync(queue, cancellationSource, signalerTask);
        }
    }

    [TestMethod]
    [RegressionTestCase("concurrent-delivery", "Concurrent priority workers deliver every request exactly once", "Twenty thousand mixed-priority requests processed by one thousand workers must complete without loss or duplication.")]
    [Timeout(15000)]
    public async Task ThousandMixedPriorityWorkers_DeliverEveryRequestExactlyOnce()
    {
        const int workerCount = 1000;
        const int requestCount = 20000;
        var queue = new ConcurrentPriQueue<int>(
            Options.Create(new ProxyConfig { MaxQueueLength = 1000 }),
            NullLogger<ConcurrentPriQueue<int>>.Instance);
        using var signalerCancellationSource = new CancellationTokenSource();
        using var producerCancellationSource = new CancellationTokenSource();
        var signalerTask = queue.SignalWorker(signalerCancellationSource.Token);
        var deliveryCounts = new int[requestCount];
        var consumed = 0;
        var allConsumed = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);

        var workerTasks = Enumerable.Range(0, workerCount)
            .Select(workerIndex => ConsumeUntilCanceledAsync(GetWorkerPriority(workerIndex)))
            .ToArray();
        var producerTask = Task.Factory.StartNew(
            Produce,
            producerCancellationSource.Token,
            TaskCreationOptions.LongRunning,
            TaskScheduler.Default);

        try
        {
            await producerTask.WaitAsync(TimeSpan.FromSeconds(10));
            await allConsumed.Task.WaitAsync(TimeSpan.FromSeconds(10));

            Assert.AreEqual(requestCount, Volatile.Read(ref consumed));
            Assert.AreEqual(0, queue.thrdSafeCount);
            Assert.IsTrue(deliveryCounts.All(count => count == 1));
        }
        finally
        {
            producerCancellationSource.Cancel();
            signalerCancellationSource.Cancel();
            await signalerTask.WaitAsync(TimeSpan.FromSeconds(2));
            await Task.WhenAll(workerTasks).WaitAsync(TimeSpan.FromSeconds(2));
        }

        async Task ConsumeUntilCanceledAsync(int preferredPriority)
        {
            try
            {
                while (true)
                {
                    var request = await queue.DequeueAsync(preferredPriority);
                    Interlocked.Increment(ref deliveryCounts[request]);
                    if (Interlocked.Increment(ref consumed) == requestCount)
                    {
                        allConsumed.TrySetResult();
                    }
                }
            }
            catch (TaskCanceledException) when (signalerCancellationSource.IsCancellationRequested)
            {
            }
        }

        void Produce()
        {
            var timestamp = DateTime.UtcNow;
            for (var request = 0; request < requestCount; request++)
            {
                producerCancellationSource.Token.ThrowIfCancellationRequested();
                var priority = request % 4 + 1;
                var priority2 = request & 1;
                while (!queue.Enqueue(request, priority, priority2, timestamp.AddTicks(request)))
                {
                    producerCancellationSource.Token.ThrowIfCancellationRequested();
                    Thread.Yield();
                }
            }
        }
    }

    private static ConcurrentPriQueue<Guid> CreateQueue(int maxQueueLength = 10)
    {
        return new ConcurrentPriQueue<Guid>(
            Options.Create(new ProxyConfig { MaxQueueLength = maxQueueLength }),
            NullLogger<ConcurrentPriQueue<Guid>>.Instance);
    }

    private static int GetWorkerPriority(int workerIndex)
    {
        if (workerIndex == 0)
        {
            return 0;
        }

        var prioritySlot = (workerIndex - 1) % 5;
        return prioritySlot switch
        {
            0 => 1,
            1 => 2,
            2 => 3,
            3 => 4,
            _ => Constants.AnyPriority
        };
    }

    private static async Task DrainAndStopAsync(
        ConcurrentPriQueue<Guid> queue,
        CancellationTokenSource cancellationSource,
        Task signalerTask)
    {
        while (queue.thrdSafeCount > 0)
        {
            await queue.DequeueAsync(Constants.AnyPriority).WaitAsync(TimeSpan.FromSeconds(1));
        }

        cancellationSource.Cancel();
        await signalerTask.WaitAsync(TimeSpan.FromSeconds(1));
    }
}

[TestClass]
[DoNotParallelize]
public sealed class UserPriorityTests : IRegressionTestMetadata
{
    public IReadOnlyDictionary<string, RegressionFeature> RegressionFeatures { get; } =
        new Dictionary<string, RegressionFeature>
        {
            ["per-user-fairness"] = new(
                "Queueing & Scheduling",
                "Per-user fairness",
                "Prevents one high-volume user from monopolizing shared worker capacity.")
        };

    [TestMethod]
    [RegressionTestCase(
        "per-user-fairness",
        "Heavy users receive a lower scheduling boost",
        "A user owning more than its fair share of active requests must lose the boost granted to lighter users.")]
    public void BoostIndicator_UsesUsersShareOfActiveRequests()
    {
        var userPriority = CreateUserPriority(threshold: 0.5f);

        Assert.IsTrue(userPriority.boostIndicator("new-user", out var newUserShare));
        Assert.AreEqual(0.0f, newUserShare);

        userPriority.addRequest("user-a");
        Assert.IsFalse(userPriority.boostIndicator("user-a", out var onlyUserShare));
        Assert.AreEqual(1.0f, onlyUserShare);

        userPriority.addRequest("user-b");
        Assert.IsFalse(userPriority.boostIndicator("user-a", out var equalShare));
        Assert.AreEqual(0.5f, equalShare);

        userPriority.addRequest("user-b");
        Assert.IsTrue(userPriority.boostIndicator("user-a", out var underrepresentedShare));
        Assert.AreEqual(1.0f / 3.0f, underrepresentedShare, 0.0001f);
        Assert.IsFalse(userPriority.boostIndicator("user-b", out var dominantShare));
        Assert.AreEqual(2.0f / 3.0f, dominantShare, 0.0001f);
    }

    [TestMethod]
    [RegressionTestCase(
        "per-user-fairness",
        "User request accounting remains consistent",
        "Adding and removing requests must keep per-user and global active counts accurate for fair scheduling.")]
    public void AddAndRemoveRequest_MaintainsUserAndTotalState()
    {
        var userPriority = CreateUserPriority(threshold: 0.5f);
        var explicitRequest = Guid.NewGuid();
        userPriority.addRequest(explicitRequest, "user-a");
        var generatedRequest = userPriority.addRequest("user-a");

        Assert.AreEqual(2, userPriority.getTotalRequests());
        CollectionAssert.AreEquivalent(
            new[] { explicitRequest, generatedRequest },
            userPriority.getUserRequests("user-a").ToArray());
        CollectionAssert.Contains(userPriority.GetAllUserIds().ToArray(), "user-a");

        Assert.IsTrue(userPriority.removeRequest("user-a", generatedRequest));
        Assert.IsFalse(userPriority.removeRequest("user-a", generatedRequest));
        Assert.AreEqual(1, userPriority.getTotalRequests());

        Assert.IsTrue(userPriority.removeRequest("user-a", explicitRequest));
        Assert.AreEqual(0, userPriority.getTotalRequests());
        Assert.AreEqual(0, userPriority.getUserRequests("user-a").Count());
        CollectionAssert.DoesNotContain(userPriority.GetAllUserIds().ToArray(), "user-a");
    }

    private static UserPriority CreateUserPriority(float threshold)
    {
        return new UserPriority(
            Options.Create(new ProxyConfig { UserPriorityThreshold = threshold }),
            NullLogger<Server>.Instance);
    }
